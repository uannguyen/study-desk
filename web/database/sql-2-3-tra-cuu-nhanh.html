2. Tạo và xóa bảng

Tạo bảng users
CREATE TABLE users (
  id         BIGSERIAL PRIMARY KEY,
  email      TEXT NOT NULL UNIQUE,
  name       TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
Tạo bảng có khóa ngoại (wallets)
CREATE TABLE wallets (
  id         BIGSERIAL PRIMARY KEY,
  user_id    BIGINT NOT NULL REFERENCES users(id),
  balance    NUMERIC(18,2) NOT NULL DEFAULT 0 CHECK (balance >= 0),
  currency   TEXT NOT NULL DEFAULT 'VND' CHECK (currency IN ('VND', 'USD')),
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
Tạo bảng transactions
CREATE TABLE transactions (
  id             BIGSERIAL PRIMARY KEY,
  from_wallet_id BIGINT NOT NULL REFERENCES wallets(id),
  to_wallet_id   BIGINT NOT NULL REFERENCES wallets(id),
  amount         NUMERIC(18,2) NOT NULL,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);
Tạo bảng, bỏ qua nếu đã có
CREATE TABLE IF NOT EXISTS users ( ... );
Đổi tên bảng
ALTER TABLE users RENAME TO members;
Xóa bảngmất cả dữ liệu
DROP TABLE IF EXISTS transactions;
Xóa bảng kèm mọi thứ phụ thuộc nóxóa luôn FK của bảng khác trỏ vào
DROP TABLE IF EXISTS users CASCADE;

3. Sửa cột (ALTER TABLE) phần lớn không ghi lại dòng cũ, trừ đổi kiểu cột

Thêm cộttức thì, dòng cũ = NULL
ALTER TABLE users ADD COLUMN phone TEXT;
Thêm cột, bỏ qua nếu đã có
ALTER TABLE users ADD COLUMN IF NOT EXISTS name TEXT;
Thêm cột có giá trị mặc địnhtức thì từ PG11, dòng cũ nhận default
ALTER TABLE users ADD COLUMN is_active BOOLEAN NOT NULL DEFAULT true;
Thêm nhiều cột một lần
ALTER TABLE users
  ADD COLUMN age SMALLINT,
  ADD COLUMN sex TEXT;
Xóa cột
ALTER TABLE users DROP COLUMN phone;
Đổi tên cột
ALTER TABLE wallets RENAME COLUMN balance TO amount;
Đổi kiểu cộtghi lại cả bảng nếu phải đổi dữ liệu
ALTER TABLE wallets ALTER COLUMN balance TYPE NUMERIC(20,2);
Đổi kiểu cột khi Postgres không tự ép đượcvd. TEXT → INTEGER
ALTER TABLE users ALTER COLUMN phone TYPE BIGINT USING phone::bigint;
Bắt buộc cột phải có giá trịlỗi nếu còn dòng NULL, điền trước
UPDATE users SET phone = '' WHERE phone IS NULL;
ALTER TABLE users ALTER COLUMN phone SET NOT NULL;
Cho phép cột NULL trở lại
ALTER TABLE users ALTER COLUMN phone DROP NOT NULL;
Đặt / bỏ giá trị mặc địnhchỉ áp cho dòng thêm sau này
ALTER TABLE users ALTER COLUMN phone SET DEFAULT '';
ALTER TABLE users ALTER COLUMN phone DROP DEFAULT;

4. Ràng buộc và index

Thêm UNIQUE cho cột
ALTER TABLE users ADD CONSTRAINT uq_users_email UNIQUE (email);
Thêm CHECK
ALTER TABLE transactions ADD CONSTRAINT chk_amount CHECK (amount > 0);
Thêm khóa ngoại (FK)
ALTER TABLE wallets ADD CONSTRAINT fk_wallets_user
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT;
Xóa ràng buộcxem tên bằng \d tên_bảng
ALTER TABLE transactions DROP CONSTRAINT chk_amount;
Tạo index
CREATE INDEX idx_tx_from ON transactions (from_wallet_id);
Tạo index nhiều cộtcột lọc bằng = đứng trước
CREATE INDEX idx_tx_from_time ON transactions (from_wallet_id, created_at);
Tạo index không khóa ghikhông chạy được trong BEGIN
CREATE INDEX CONCURRENTLY idx_wallets_user ON wallets (user_id);
Tạo unique index
CREATE UNIQUE INDEX uq_wallets_user ON wallets (user_id);
Xóa index
DROP INDEX IF EXISTS idx_tx_from;

5. Thêm dữ liệu (INSERT)

Thêm một dòng
INSERT INTO users (name, email) VALUES ('An', 'an@lab.vn');
Thêm nhiều dòng
INSERT INTO users (name, email) VALUES
  ('Bình', 'binh@lab.vn'),
  ('Chi',  'chi@lab.vn');
Thêm rồi lấy lại id vừa tạo
INSERT INTO users (name, email) VALUES ('Dũng', 'dung@lab.vn') RETURNING id;
Seed 10k users
INSERT INTO users (name, email)
SELECT 'user ' || g, 'u' || g || '@lab.vn'
FROM generate_series(1, 10000) AS g;
Mỗi user một víINSERT … SELECT
INSERT INTO wallets (user_id, balance, currency)
SELECT id,
       (random() * 1000000)::numeric(18,2),
       (ARRAY['VND','USD'])[1 + floor(random() * 2)::int]
FROM users;
Seed 100k giao dịch chuyển tiềnbốc id ví có thật, chạy đúng cả khi id có lỗ
WITH w AS (
  SELECT array_agg(id ORDER BY id) AS ids, count(*)::int AS n FROM wallets
),
s AS (
  SELECT floor(power(random(), 3) * w.n)::int + 1 AS f,  -- ví gửi, lệch về đầu mảng
         floor(random() * (w.n - 1))::int         AS k   -- ví nhận luôn khác ví gửi
  FROM w, generate_series(1, 100000)
)
INSERT INTO transactions (from_wallet_id, to_wallet_id, amount, created_at)
SELECT w.ids[s.f],
       w.ids[(s.f + s.k) % w.n + 1],
       (1 + random() * 499999)::numeric(18,2),
       now() - random() * interval '365 days'
FROM s, w;
Thêm, nếu trùng thì cập nhật (upsert)
INSERT INTO users (email, name) VALUES ('an@lab.vn', 'An mới')
ON CONFLICT (email) DO UPDATE SET name = EXCLUDED.name;
Thêm, nếu trùng thì bỏ qua
INSERT INTO users (email, name) VALUES ('an@lab.vn', 'An')
ON CONFLICT (email) DO NOTHING;

6. Sửa và xóa dữ liệu luôn chạy SELECT với cùng WHERE trước

Sửa một dòng
UPDATE users SET name = 'An Nguyễn' WHERE id = 1;
Cộng / trừ giá trị
UPDATE wallets SET balance = balance - 50000 WHERE id = 1;
Sửa dựa trên bảng khác (UPDATE … FROM)
UPDATE wallets w SET balance = 0
FROM users u
WHERE w.user_id = u.id AND u.email LIKE '%@test.vn';
Sửa rồi xem lại dòng đã sửa
UPDATE wallets SET balance = balance + 1000 WHERE id = 1 RETURNING id, balance;
Xóa dòng
DELETE FROM transactions WHERE created_at < now() - interval '1 year';
Xóa sạch bảng, reset id về 1không hoàn tác được ngoài transaction
TRUNCATE transactions, wallets, users RESTART IDENTITY;

7. Đọc dữ liệu (SELECT)

Lấy vài dòng xem thử
SELECT * FROM users LIMIT 10;
Lọc theo điều kiện
SELECT id, name FROM users WHERE created_at >= '2026-01-01';
Tìm theo chuỗi (không phân biệt hoa thường)
SELECT * FROM users WHERE name ILIKE '%an%';
Lọc theo danh sách / khoảng
SELECT * FROM wallets WHERE user_id IN (1, 2, 3);
SELECT * FROM wallets WHERE balance BETWEEN 100000 AND 500000;
Lọc giá trị rỗngkhông dùng = NULL
SELECT * FROM users WHERE phone IS NULL;
Sắp xếp và phân trang
SELECT * FROM transactions ORDER BY created_at DESC LIMIT 20 OFFSET 40;
Đếm số dòng
SELECT count(*) FROM transactions;
Lấy giá trị không trùng
SELECT DISTINCT currency FROM wallets;
Gom nhóm và tính tổng
SELECT currency, count(*), sum(balance)
FROM wallets GROUP BY currency;
Lọc sau khi gom nhóm (HAVING)
SELECT from_wallet_id, count(*) AS n
FROM transactions GROUP BY from_wallet_id
HAVING count(*) > 5 ORDER BY n DESC;
Nối hai bảng (INNER JOIN)
SELECT u.name, w.balance
FROM users u JOIN wallets w ON w.user_id = u.id;
Nối, giữ cả dòng không khớp (LEFT JOIN)điều kiện bảng phải đặt ở ON
SELECT u.id, w.id AS wallet_id
FROM users u LEFT JOIN wallets w ON w.user_id = u.id
WHERE w.id IS NULL;   -- user chưa có ví
Tìm dòng có liên quan ở bảng khác (EXISTS)
SELECT * FROM wallets w
WHERE EXISTS (SELECT 1 FROM transactions t WHERE t.from_wallet_id = w.id);  -- ví đã từng gửi tiền
Đặt tên cho query con (CTE)
WITH big AS (
  SELECT from_wallet_id, sum(amount) AS total FROM transactions GROUP BY from_wallet_id
)
SELECT * FROM big WHERE total > 1000000;

8. Transaction và lock

Chạy nhiều lệnh như một khối
BEGIN;
UPDATE wallets SET balance = balance - 100 WHERE id = 1;
UPDATE wallets SET balance = balance + 100 WHERE id = 2;
COMMIT;   -- hoặc ROLLBACK; để hủy
Khóa dòng trước khi sửalock theo id tăng dần để tránh deadlock
BEGIN;
SELECT * FROM wallets WHERE id IN (1, 2) ORDER BY id FOR UPDATE;
-- UPDATE ...
COMMIT;
Hủy một phần trong transaction
SAVEPOINT sp1;
-- lệnh lỗi ...
ROLLBACK TO SAVEPOINT sp1;
Đổi isolation level
BEGIN ISOLATION LEVEL REPEATABLE READ;
Xem ai đang chờ lock
SELECT pid, state, wait_event_type, query FROM pg_stat_activity
WHERE wait_event_type = 'Lock';

9. Kiểm tra hiệu năng

Xem kế hoạch chạy thậtANALYZE chạy thật, cẩn thận với UPDATE/DELETE
EXPLAIN ANALYZE SELECT * FROM transactions WHERE from_wallet_id = 42;
Xem thêm số block đọc từ cache / đĩa
EXPLAIN (ANALYZE, BUFFERS) SELECT * FROM transactions WHERE from_wallet_id = 42;
Cập nhật thống kê cho plannerchạy sau khi seed nhiều dữ liệu
ANALYZE transactions;
Xem dung lượng bảng (kèm index)
SELECT pg_size_pretty(pg_total_relation_size('transactions'));

Không có dòng nào khớp. Thử từ khác, ví dụ "cột", "index", "xóa".

Cú pháp theo Postgres 11 trở lên. Phần giải thích vì sao, bẫy và câu hỏi phỏng vấn nằm ở các bản Cornell 2.0 đến 2.2.
